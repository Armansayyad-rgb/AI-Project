# Runtime dependencies for AI-Project.
# Keep bounds compatible with currently published stable PyPI releases so a
# fresh clone can install reproducibly in Docker and local Python environments.

tokenizers>=0.22.2,<=0.23.1

# Numerical utilities used by models and tokenizers.

# Used by src/download_corpus.py to pull the WikiText corpus from HF Hub.

# Web UI. The current UI uses Textbox(show_copy_button=...), which is part of
# the Gradio 4.44.1 API. Pin it so fresh Docker builds do not break on newer
# incompatible Gradio releases.

# Document upload pipeline.
